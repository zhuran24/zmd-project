/-- Pairwise disjointness for a finite family of occupied cell sets. -/
def PairwiseDisjoint {ι α : Type u} (F : ι → Finset α) : Prop :=
  ∀ ⦃i j : ι⦄, i ≠ j → Disjoint (F i) (F j)

/-- Boolean value of the project objective `max_lex(area, min_side)`, written as a pair. -/
abbrev LexValue := Nat × Nat

/-- `a` is lexicographically no better than `b`: area first, then min_side. -/
def LexLe (a b : LexValue) : Prop :=
  a.1 < b.1 ∨ (a.1 = b.1 ∧ a.2 ≤ b.2)

/-- Strict lex improvement: area first, then min_side. -/
def LexLt (a b : LexValue) : Prop :=
  a.1 < b.1 ∨ (a.1 = b.1 ∧ a.2 < b.2)

/-- `x` is a lexicographic maximum of the feasible set. -/
def IsLexMax {Cand : Type u} (Feasible : Cand → Prop) (value : Cand → LexValue)
    (x : Cand) : Prop :=
  Feasible x ∧ ∀ y : Cand, Feasible y → LexLe (value y) (value x)

/-- The search set after frontier pruning with a certified witness kept as incumbent. -/
def FrontierKept {Cand : Type u} (Feasible : Cand → Prop) (value : Cand → LexValue)
    (witness cand : Cand) : Prop :=
  Feasible cand ∧ (cand = witness ∨ LexLt (value witness) (value cand))

/-- TP7-S T1: per-commodity conservation at non-terminal nodes. -/
def TP7_T1 {Node Edge Commodity : Type u} [DecidableEq Edge]
    (InEdges OutEdges : Node → Finset Edge)
    (NonTerminal : Node → Commodity → Prop)
    (φ : Edge → Commodity → ℚ) : Prop :=
  ∀ n k, NonTerminal n k →
    (∑ e in InEdges n, φ e k) = (∑ e in OutEdges n, φ e k)

/-- TP7-S T2: selected physical states obey aggregate belt capacity. -/
def TP7_T2 {PhysState Edge Commodity : Type u}
    (SelectedPhysical : PhysState → Prop)
    (Through : (Edge → Commodity → ℚ) → PhysState → ℚ)
    (beltCapacityPerTick : ℚ)
    (φ : Edge → Commodity → ℚ) : Prop :=
  ∀ s, SelectedPhysical s → Through φ s ≤ beltCapacityPerTick

/-- TP7-S T3: port bounds, route-visible terminal equality, and zero-throughput rejection cases. -/
def TP7_T3 {Port Edge Commodity : Type u}
    (RouteVisible KnownPort UniquePort IncidentSelectedState : Port → Prop)
    (PortMax : Port → ℚ)
    (TerminalArcFlow : (Edge → Commodity → ℚ) → Port → ℚ)
    (φ : Edge → Commodity → ℚ) (r : Port → ℚ) : Prop :=
  ∀ p, 0 ≤ r p ∧ r p ≤ PortMax p ∧
    (RouteVisible p → r p = TerminalArcFlow φ p) ∧
    ((¬ KnownPort p ∨ ¬ UniquePort p ∨ ¬ IncidentSelectedState p) → r p = 0)

/-- TP7-S T4: machine input/output coupling through utilization. -/
def TP7_T4 {Machine Commodity Port : Type u} [DecidableEq Port]
    (InPorts OutPorts : Machine → Commodity → Finset Port)
    (InputQty OutputQty : Machine → Commodity → ℚ)
    (Tpc : Machine → ℚ)
    (r : Port → ℚ) (u : Machine → ℚ) : Prop :=
  ∀ i k, 0 < Tpc i ∧
    (∑ p in InPorts i k, r p) = u i * InputQty i k / Tpc i ∧
    (∑ p in OutPorts i k, r p) = u i * OutputQty i k / Tpc i

/-- TP7-S T5: production target rates are exact equalities. -/
def TP7_T5 {Target Port : Type u} [DecidableEq Port]
    (TargetOutputPorts : Target → Finset Port)
    (TargetRate : Target → ℚ)
    (r : Port → ℚ) : Prop :=
  ∀ t, (∑ p in TargetOutputPorts t, r p) = TargetRate t

/-- TP7-S T6: routing-free sink balance and external-boundary-only injection. -/
def TP7_T6 {Commodity Port : Type u} [DecidableEq Port]
    (RoutingFreeSink : Commodity → Prop)
    (ProductionPorts VirtualSinkSlots : Commodity → Finset Port)
    (ExternalInjectionPort SourceKindExternalBoundary : Port → Prop)

/-- Full 0/1 equality-key nogood constraint: at least one bit differs from the recorded assignment. -/
def FullEqKeyConstraint {Var : Type u} [Fintype Var]
    (key cand : Var → Bool) : Prop :=
  1 ≤ ∑ v : Var, if cand v = key v then (0 : Nat) else 1

/-- The assignments excluded by the full equality-key nogood. -/
def FullEqKeyExcluded {Var : Type u} [Fintype Var]
    (key cand : Var → Bool) : Prop :=
  ¬ FullEqKeyConstraint key cand

/-- Selected-only nogood: only the variables set to 1 in the recorded solution appear. -/
def SelectedOnlyConstraint {Var : Type u} [Fintype Var]
    (key cand : Var → Bool) : Prop :=
  1 ≤ ∑ v : Var, if key v = true then (if cand v = true then (0 : Nat) else 1) else 0

/-- The assignments excluded by selected-only nogood. -/
def SelectedOnlyKeyExcluded {Var : Type u} [Fintype Var]
    (key cand : Var → Bool) : Prop :=
  ¬ SelectedOnlyConstraint key cand
